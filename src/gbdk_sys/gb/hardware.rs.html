3/// Volatile pointer to a memory-mapped hardware register.
4#[derive(Clone, Copy)]
5pub struct Reg(pub *mut u8);
6
7impl Reg {
8    #[inline(always)]
9    pub unsafe fn read(self) -> u8 {
10        unsafe { core::ptr::read_volatile(self.0) }
11    }
12    #[inline(always)]
13    pub unsafe fn write(self, val: u8) {
14        unsafe { core::ptr::write_volatile(self.0, val) }
15    }
16}
