5pub const fn rgb(r: u8, g: u8, b: u8) -> PaletteColor {
6    ((b as u16 & 0x1f) << 10) | ((g as u16 & 0x1f) << 5) | (r as u16 & 0x1f)
7}
8
9pub const fn rgb8(r: u8, g: u8, b: u8) -> PaletteColor {
10    rgb(r >> 3, g >> 3, b >> 3)
11}
12
13pub const fn rgbhtml(rgb24: u32) -> PaletteColor {
14    rgb8((rgb24 >> 16) as u8, (rgb24 >> 8) as u8, rgb24 as u8)
15}
