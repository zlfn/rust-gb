5unsafe extern "sdcccall-0" {
6    pub fn reverse(s: *mut c_char) -> *mut c_char;
7    pub fn strcpy(dest: *mut c_char, src: *const c_char) -> *mut c_char;
8    pub fn strcmp(s1: *const c_char, s2: *const c_char) -> c_int;

11// strlen/memcmp/memset/memcpy/memmove are runtime symbols that compiler_builtins
12// already provides. The linker resolves them to those extern "C" (sdcccall(1))
13// definitions, so we must declare them with the same ABI and canonical signature.
14// The sdcccall(0) variants in gbdk's libsm83.a use `__sdcc`-suffixed names and are
15// never linked.
16unsafe extern "C" {
17    pub fn strcat(s1: *mut c_char, s2: *const c_char) -> *mut c_char;
18    pub fn strncat(s1: *mut c_char, s2: *const c_char, n: c_int) -> *mut c_char;
19    pub fn strncmp(s1: *const c_char, s2: *const c_char, n: c_int) -> c_int;
20    pub fn strncpy(s1: *mut c_char, s2: *const c_char, n: c_int) -> *mut c_char;
21
22    pub fn strlen(s: *const c_char) -> usize;
23    pub fn memcmp(buf1: *const c_void, buf2: *const c_void, count: usize) -> c_int;
24
25    #[deprecated(note = "use core::ptr::write_bytes")]
26    pub fn memset(s: *mut c_void, c: c_int, n: usize) -> *mut c_void;
27    #[deprecated(note = "use core::ptr::copy_nonoverlapping")]
28    pub fn memcpy(dest: *mut c_void, src: *const c_void, len: usize) -> *mut c_void;
29    #[deprecated(note = "use core::ptr::copy")]
30    pub fn memmove(dest: *mut c_void, src: *const c_void, n: usize) -> *mut c_void;
31}
