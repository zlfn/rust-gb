4//! # Run-time support for Game Boy (SM83)
5//!
6//! This crate provides startup code that LLVM-Z80 and other crates in Rust-GB expect.
7//!
8//! Depending on this crate links the startup code in `rrt0.s`
9//! and makes the linker script `gb.ld` available to the ROM build pipeline.
10
11core::arch::global_asm!(include_str!("rrt0.s"));

27/// Whether this program was built for CGB double speed mode.
28///
29/// The `cgb-double-speed` feature sets it, and the startup switches before
30/// [`entry`](macro@crate::entry) hands over.
31///
32/// Other crates can read this value to determine the current Game Boy's clock speed.
33pub const DOUBLE_SPEED: bool = cfg!(feature = "cgb-double-speed");
34
35/// Switch into CGB double speed mode.
36///
37/// This should be called once, before the program runs.
38/// [`macro@entry`] runs this automatically depending on the feature flag.
39#[doc(hidden)]
40#[inline]
41pub fn __enter_double_speed() {
42    // If the code below runs on the original Game Boy, it will freeze.
43    // TODO: Prevent this at compile time in cargo-gb?
44    if !DOUBLE_SPEED || crate::boot::BOOT_A != 0x11 {
45        return;
46    }
47    unsafe {
48        core::arch::asm!(
49            // The following procedure follows Pan Docs's speed switching process.
50            // https://gbdev.io/pandocs/CGB_Registers.html#ff4d--key1spd-cgb-mode-only-prepare-speed-switch
51            "xor a",
52            "ldh ($ff), a",     // IE
53            "ld a, $30",
54            "ldh ($00), a",     // JOYP
55            "ld a, $01",
56            "ldh ($4d), a",     // KEY1
57            // TODO: Replace this if `stop` mnemonic is supported in LLVM-Z80.
58            ".byte 0x10, 0x00", // `stop`
59            out("a") _,
60            options(nostack),
61        );
62    }
63}
